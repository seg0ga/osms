4) Взял N, равное 250 битам. Проделал п.1-3
``` cpp
#include <iostream>
#include <cstdio>

using namespace std;

void packet(int data[], int N, int G[], int kadr[]){
    int copy[512];
    for (int i=0;i<N+6;i++){
        if (i>=N){copy[i]=0;}
        else copy[i]=data[i];
    };

    for (int i=0;i<N;i++){
        if (copy[i]==1){
            for (int j=0;j<7;j++){
                copy[i+j] ^= G[j];}}}

    int crc[6];

    printf("CRC:");
    for (int i=0; i<6;i++){
        crc[i]=copy[N+i];
        cout << crc[i];
    }
    cout <<endl;

    for (int i=0;i<N+6;i++){
        if (i>=N){kadr[i]=crc[i-N];}
        else kadr[i]=data[i];
    };

    int check[512];
    for (int i=0;i<N+6;i++) check[i]=kadr[i];
    for (int i=0;i<N;i++){
        if (check[i]==1){
            for (int j=0;j<7;j++) check[i+j] ^=G[j];}}

    int error=0;
    printf("Остаток при проверке: ");
    for (int i=0;i<6;i++){
        cout <<check[N+i];
        if (check[N+i]==1) error=1;
    }
    cout <<endl;
    if (error) printf("Ошибка обнаружена :(\n");
    else printf("Ошибок нет :)\n");
}

int main(){
    int G[7]={1,1,1,1,1,1,1};
    int N=20+8;
    int data[40];
    int kadr[40];

    printf("Пакет данных длиной 28 бит\n\n");

    printf("Введите 28 бит через пробел: ");
    for (int i=0;i<N;i++){
        cin >> data[i];}
    
    packet(data,N,G,kadr);

    int N250=250;

        data250[i]=rand()%2;
        printf("%d",data250[i]);}
    cout <<endl;
    
    packet(data250,N250,G,kadr250);
}
```

## Заключение
В рамках практической работы была написана программа на языке C++ для вычисления циклического избыточного кода CRC и проверки целостности передаваемых пакетов данных. 

На первом этапе алгоритм был проверен на пакете длиной 28 бит, а затем на случайно сгенерированной последовательности из 250 бит. В обоих случаях при отсутствии искажений повторное деление принятого кадра на порождающий полином давало нулевой остаток, что подтверждает правильность вычисления и проверки кода. 

На втором этапе был проведен эксперимент с поочередным искажением каждого бита в кадре длиной 256 бит. Приемник зафиксировал ненулевой остаток во всех 256 случаях, следовательно,  ни одна ошибка не была пропущена. 

Работа показала, что метод CRC позволяет эффективно выявлять одиночные ошибки в канале связи с помощью простых побитовых операций XOR. При этом объем избыточной информации остается небольшим и зависит только от степени порождающего полинома, а не от длины самого пакета.
